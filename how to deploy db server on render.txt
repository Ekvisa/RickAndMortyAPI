1) Инициализировать проект:
npm init -y
В результате появится package.json.

2) Установить json-server локально:
npm install json-server --save-dev
Это создаст node_modules/ и package-lock.json, а в package.json появится запись в devDependencies.
Проверка установки:
npx json-server --version
# или
npm list json-server

3) Создай и заполни своими данными db.json (положи в корень проекта):
{
  "characters": [],
  "locations": [],
  "episodes": []
}

4) Рекомендуемый способ: запуск через server.js (чтобы учитывать переменную окружения PORT)
Создай server.js в корне — это лучший способ, потому что Render задаёт порт через PORT, и так же удобно локально менять порт.
// server.js
const jsonServer = require("json-server");
const server = jsonServer.create();
const router = jsonServer.router("db.json");
const middlewares = jsonServer.defaults();
const port = process.env.PORT || 3001; // fallback локально 3001
server.use(middlewares);
server.use(router);
server.listen(port, () => {
  console.log(`JSON Server is running on port ${port}`);
});


5) Обновить package.json — скрипт старта.
В package.json добавь/замени секцию scripts:
"scripts": {
  "start": "node server.js",
  "dev": "node server.js"
}

6)Теперь локально запускаем:
npm start
или с указанным портом (Windows cmd):
PORT=3002 npm start

7) Как это будет работать на Render:
Render при деплое клонирует репозиторий и выполняет npm install. Когда он запускает сервис, он выставит переменную окружения PORT. Поскольку наш server.js читает process.env.PORT, json-server запустится на нужном порту автоматически — никаких дополнительных хитростей не нужно.

8) Быстрое резюме команд:
npm init -y
npm install json-server --save-dev
# создать db.json и server.js (см. выше)
# в package.json: "start": "node server.js"
npm start
# или с другим портом
PORT=3002 npm start
